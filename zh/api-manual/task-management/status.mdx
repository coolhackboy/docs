# 获取任务状态

查询图像、视频、音频、3D 和内容审核任务的执行状态，并在完成后获取结果。

## 接口

```http
GET https://api.poyo.ai/api/generate/status/{task_id}
```

## 认证

所有请求都需要 Bearer token 认证：

```bash
Authorization: Bearer YOUR_API_KEY
```

<Card title="获取 API Key" icon="key" href="https://poyo.ai/dashboard/api-key">
  在控制台生成你的 API Key
</Card>

## 参数

### 路径参数

| 参数 | 类型 | 必填 | 说明 |
|-----------|------|----------|-------------|
| `task_id` | string | 是 | 提交接口返回的唯一任务标识 |

## 响应

### 成功响应（200）

```json
{
  "code": 200,
  "data": {
    "task_id": "task-unified-1757165031-uyujaw3d",
    "status": "finished",
    "credits_amount": 2,
    "files": [
      {
        "file_url": "https://storage.poyo.ai/generated/image-abc123.jpg",
        "file_type": "image"
      }
    ],
    "created_time": "2025-11-12T10:30:00",
    "progress": 100,
    "error_message": null
  }
}
```

### 响应字段

| 字段 | 类型 | 说明 |
|-------|------|-------------|
| `code` | integer | HTTP 状态码（成功为 200） |
| `data.task_id` | string | 唯一任务标识 |
| `data.status` | string | 任务状态：`not_started`、`running`、`finished`、`failed` |
| `data.credits_amount` | number | 该任务扣除的积分 |
| `data.output` | object | 可选的结构化模型输出。Omni Moderation 在这里返回完整的 OpenAI moderation 响应，同时 `files: []`、`credits_amount: 0` |
| `data.files` | array | 生成的媒体文件（状态为 `finished` 时） |
| `data.files[].file_url` | string | 生成文件的直接 URL |
| `data.files[].file_type` | string | 生成文件的类型：`image`、`video`、`audio`、`3d` 或 `other` |
| `data.files[].label` | string | 可选的文件标签。常见于 3D 输出，例如 `model_glb` 或 `thumbnail` |
| `data.files[].format` | string | 可选的文件格式，例如 `glb`、`fbx`、`obj`、`png` 或 `mp4` |
| `data.files[].content_type` | string | 可选的生成文件 MIME 类型 |
| `data.files[].file_name` | string | 可选的文件名 |
| `data.files[].file_size` | integer | 可选的文件大小（字节） |
| `data.files[]` 结构化字段 | object | 部分音频工作流返回 `voice_id`、`voice_status`、`is_available` 或 `origin_data` 等字段，而不是单个媒体 URL |
| `data.created_time` | string | 任务创建时间，ISO 8601 时间戳 |
| `data.progress` | integer | 完成百分比（0-100） |
| `data.error_message` | string | 错误描述（状态为 `failed` 时） |

对于 [Omni Moderation](/zh/api-manual/moderation-series/omni-moderation)，在状态变为 `finished` 后检查 `data.output.results[].flagged`。`flagged: true` 表示审核已完成并检测到相关内容，不会使任务变为 `failed`。Omni Moderation 异步运行，支持通过 `callback_url` 回调，不支持流式输出。其回调体与这里返回的任务数据对象相同，有 `output` 时会包含 `output`，并且 `credits_amount: 0`。

### 状态值

| 状态 | 说明 |
|--------|-------------|
| `not_started` | 任务已排队，等待处理 |
| `running` | 任务正在生成 |
| `finished` | 任务成功完成，结果在 `files` 或 `output` 中 |
| `failed` | 任务失败，错误详情在 `error_message` 中 |

### Go

```go
package main

import (
    "encoding/json"
    "fmt"
    "io"
    "net/http"
)

const (
    APIKey = "your-api-key-here"
    BaseURL = "https://api.poyo.ai"
)

type APIResponse struct {
    Code int      `json:"code"`
    Data TaskData `json:"data"`
}

type TaskData struct {
    TaskID       string      `json:"task_id"`
    Status       string      `json:"status"`
    Files        []map[string]any `json:"files"`
    CreatedTime  string      `json:"created_time"`
    Progress     int         `json:"progress"`
    CreditsAmount float64     `json:"credits_amount"`
    ErrorMessage *string     `json:"error_message"`
}

func main() {
    taskID := "task-unified-1757165031-uyujaw3d"

    req, _ := http.NewRequest(
        "GET",
        fmt.Sprintf("%s/api/generate/status/%s", BaseURL, taskID),
        nil,
    )
    req.Header.Set("Authorization", "Bearer "+APIKey)

    client := &http.Client{}
    resp, err := client.Do(req)
    if err != nil {
        panic(err)
    }
    defer resp.Body.Close()

    body, _ := io.ReadAll(resp.Body)

    var apiResp APIResponse
    json.Unmarshal(body, &apiResp)

    data := apiResp.Data
    fmt.Printf("Status: %s\n", data.Status)
    fmt.Printf("Progress: %d%%\n", data.Progress)
    fmt.Printf("Credits: %.2f\n", data.CreditsAmount)

    if data.Status == "finished" {
        for _, item := range data.Files {
            if fileURL, ok := item["file_url"].(string); ok {
                fmt.Printf("Generated file: %s\n", fileURL)
            } else {
                encoded, _ := json.Marshal(item)
                fmt.Printf("Structured result: %s\n", encoded)
            }
        }
    } else if data.Status == "failed" {
        fmt.Printf("Error: %s\n", *data.ErrorMessage)
    }
}
```

### Java

```java
import java.net.http.*;
import java.net.URI;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.JsonNode;

public class PoYoAPI {
    private static final String API_KEY = "your-api-key-here";
    private static final String BASE_URL = "https://api.poyo.ai";

    public static void main(String[] args) throws Exception {
        String taskId = "task-unified-1757165031-uyujaw3d";

        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create(BASE_URL + "/api/generate/status/" + taskId))
            .header("Authorization", "Bearer " + API_KEY)
            .GET()
            .build();

        HttpResponse<String> response = client.send(
            request,
            HttpResponse.BodyHandlers.ofString()
        );

        ObjectMapper mapper = new ObjectMapper();
        JsonNode root = mapper.readTree(response.body());
        JsonNode data = root.get("data");

        String status = data.get("status").asText();
        int progress = data.get("progress").asInt();
        double creditsAmount = data.get("credits_amount").asDouble();

        System.out.println("Status: " + status);
        System.out.println("Progress: " + progress + "%");
        System.out.println("Credits: " + creditsAmount);

        if ("finished".equals(status)) {
            JsonNode files = data.get("files");
            files.forEach(item -> {
                if (item.hasNonNull("file_url")) {
                    System.out.println("Generated file: " + item.get("file_url").asText());
                } else {
                    System.out.println("Structured result: " + item.toString());
                }
            });
        } else if ("failed".equals(status)) {
            System.out.println("Error: " + data.get("error_message").asText());
        }
    }
}
```

## 轮询最佳实践

轮询任务完成状态时，请遵循以下建议：

<Tip>
  **轮询间隔**：每 2-5 秒查询一次状态。避免更频繁地轮询，以免触发速率限制。
</Tip>

<Tip>
  **超时**：设置超时以避免无限循环。典型生成耗时：
  - 图像：15-60 秒
  - 视频：90-300 秒
</Tip>

<Tip>
  **指数退避**：遇到速率限制（429）或服务器错误（5xx）时，使用指数退避（等待 1 秒、2 秒、4 秒、8 秒……）。
</Tip>

### 完整轮询示例

```python
import requests
import time
import json

API_KEY = "your-api-key-here"
BASE_URL = "https://api.poyo.ai"

def wait_for_completion(task_id, timeout=600, poll_interval=2):
    """
    Wait for task completion with timeout

    Args:
        task_id: Task identifier
        timeout: Maximum wait time in seconds
        poll_interval: Seconds between checks

    Returns:
        Task data when complete

    Raises:
        TimeoutError: If timeout exceeded
        RuntimeError: If task failed
    """
    start_time = time.time()
    headers = {"Authorization": f"Bearer {API_KEY}"}

    while True:
        # Check timeout
        if time.time() - start_time > timeout:
            raise TimeoutError(f"Task did not complete within {timeout}s")

        # Get status
        response = requests.get(
            f"{BASE_URL}/api/generate/status/{task_id}",
            headers=headers
        )

        if response.status_code == 429:
            # Rate limited - wait longer
            time.sleep(poll_interval * 2)
            continue

        response.raise_for_status()
        result = response.json()
        data = result["data"]

        # Log progress
        elapsed = int(time.time() - start_time)
        print(f"[{elapsed}s] Status: {data['status']}, Progress: {data['progress']}%")

        # Check completion
        if data["status"] == "finished":
            print(f"✓ Task completed successfully")
            return data
        elif data["status"] == "failed":
            raise RuntimeError(f"Task failed: {data['error_message']}")

        time.sleep(poll_interval)

# Usage
try:
    task_id = "task-unified-1757165031-uyujaw3d"
    result = wait_for_completion(task_id)

    for item in result["files"]:
        if "file_url" in item:
            print(f"Download: {item['file_url']}")
        else:
            print(f"Structured result: {json.dumps(item)}")

except TimeoutError as e:
    print(f"Timeout: {e}")
except RuntimeError as e:
    print(f"Error: {e}")
```

## 错误响应

### HTTP 错误码

| 状态码 | 说明 | 处理方式 |
|------|-------------|--------|
| 401 | Unauthorized | 检查你的 API Key |
| 403 | Forbidden | 任务属于其他用户 |
| 404 | Not Found | 任务 ID 不存在 |
| 429 | Too Many Requests | 降低轮询频率，实现退避 |
| 500 | Server Error | 使用指数退避重试 |
| 502 | Bad Gateway | 服务暂时不可用，请重试 |

### 错误响应格式

```json
{
  "detail": "Task not found"
}
```
